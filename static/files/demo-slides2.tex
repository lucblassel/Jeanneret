\begin{frame}{What is Brutalism?}
\begin{itemize}
    \item Emerged in the mid-20th century
    \item Derived from \textit{béton brut} (raw concrete)
    \item Focus on functionality and material honesty
    \item Strong geometric forms and monumental scale
\end{itemize}
\end{frame}

\begin{frame}{Impact on Urban Landscapes}
\begin{itemize}
    \item Widely used in public housing and civic buildings
    \item Emphasis on social purpose and accessibility
    \item Criticized for perceived coldness and austerity
    \item Iconic examples remain influential today
\end{itemize}
\end{frame}

\begin{frame}{Legacy and Reinterpretation}
\begin{itemize}
    \item Renewed interest among contemporary architects
    \item Preservation vs. demolition debates
    \item Influence on minimalism and modern urban design
    \item Brutalism as both cultural artifact and inspiration
\end{itemize}
\end{frame}
